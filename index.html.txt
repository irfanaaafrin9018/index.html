function playSegment(file,start,end,loop=false){
  clearTimeout(stopTimer);
  bgMusic.src=file;
  bgMusic.currentTime=start;
  bgMusic.play();
  stopTimer=setTimeout(()=>{ bgMusic.pause(); if(loop) playSegment(file,start,end,true); },(end-start)*1000);
}
